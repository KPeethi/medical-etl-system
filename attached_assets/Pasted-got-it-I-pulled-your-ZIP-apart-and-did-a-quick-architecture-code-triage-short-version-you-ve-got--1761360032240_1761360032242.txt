## 3) config hash + manifest

```python
# router_service/lib/logger.py
import hashlib, json
def config_hash(cfg: dict) -> str:
    return hashlib.sha256(json.dumps(cfg, sort_keys=True).encode()).hexdigest()[:16]

def write_manifest(manifest_path, entries):
    # entries: list of {path,size,sha256}
    with open(manifest_path, "w", encoding="utf-8") as f:
        json.dump({"entries": entries}, f, indent=2)
```

# router_service/universal_router.py
import shutil, time
from datetime import datetime, timezone

def now_utc():
    return datetime.now(timezone.utc)

def ensure_disk(headroom_pct: int, path: str):
    total, used, free = shutil.disk_usage(path)
    if free/total*100 < headroom_pct:
        raise RuntimeError(f"Low disk: {free/total*100:.1f}% free")

# before copy loop
ensure_disk(self.config.get("safety",{}).get("min_free_pct",10), self.dst_root)

## 5) basic auth + security headers

```python
# review_ui/app.py
from flask import Flask, request, abort
from functools import wraps
from flask_talisman import Talisman

app = Flask(__name__)
Talisman(app, content_security_policy=None)  # sane defaults, HTTPS expected in prod

def requires_auth(fn):
    @wraps(fn)
    def _w(*a, **k):
        auth = request.authorization
        if not auth or not (auth.username==os.getenv("UI_USER") and auth.password==os.getenv("UI_PASS")):
            return abort(401)
        return fn(*a, **k)
    return _w
```
